// --- 1. 文件指纹（秒传的前提） ---

async function sha256(file: File): Promise<string> {
  const buffer = await file.arrayBuffer()
  const digest = await crypto.subtle.digest('SHA-256', buffer)
  return Array.from(new Uint8Array(digest))
    .map((b) => b.toString(16).padStart(2, '0'))
    .join('')
}

// --- 3. 切片 + 并发上传 ---

const CHUNK_SIZE = 1024 * 1024
const CONCURRENCY = 3

function sliceFile(file: File): Blob[] {
  const list: Blob[] = []
  for (let start = 0; start < file.size; start += CHUNK_SIZE) {
    //! 用 slice 而不是读进内存：Blob 是引用，不会复制数据
    list.push(file.slice(start, start + CHUNK_SIZE))
  }
  return list
}

async function uploadWithLimit(parts: Blob[], hash: string): Promise<void> {
  let cursor = 0

  const worker = async (): Promise<void> => {
    while (cursor < parts.length) {
      const index = cursor++
      const form = new FormData()
      form.append('hash', hash)
      form.append('index', String(index))
      form.append('total', String(parts.length))
      form.append('chunk', parts[index])

      //! 每个 worker 串行取任务：天然限流，不需要信号量
      await fetch('/api/upload/chunk', { method: 'POST', body: form })
    }
  }

  await Promise.all(Array.from({ length: CONCURRENCY }, worker))
}
