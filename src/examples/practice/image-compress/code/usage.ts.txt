// --- 步骤一：读文件 → Image → canvas 等比缩放 ---

async function compressImage(file: File, quality = 0.6, maxEdge = 1200): Promise<Blob | null> {
  // 文件 → objectURL → Image（比 FileReader 省一次 base64 转换）
  const url = URL.createObjectURL(file)
  const img = await loadImage(url)

  //! 等比缩放：长边压到 maxEdge 以内（手机原图 4000px 直接上传会巨慢）
  const scale = Math.min(1, maxEdge / Math.max(img.naturalWidth, img.naturalHeight))
  const canvas = document.createElement('canvas')
  canvas.width = Math.round(img.naturalWidth * scale)
  canvas.height = Math.round(img.naturalHeight * scale)

  const ctx = canvas.getContext('2d')!
  //! 缩放质量信号：让浏览器用高质量插值（缩小图片必须开）
  ctx.imageSmoothingEnabled = true
  ctx.imageSmoothingQuality = 'high'
  ctx.drawImage(img, 0, 0, canvas.width, canvas.height)

  //! 核心一行：toBlob(type, quality)——JPEG 质量 0~1，浏览器原生编码器
  const blob = await new Promise<Blob | null>((resolve) =>
    canvas.toBlob(resolve, 'image/jpeg', quality),
  )

  URL.revokeObjectURL(url) // 回收
  return blob
}

function loadImage(url: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    img.onload = () => resolve(img)
    img.onerror = reject
    img.src = url
  })
}
