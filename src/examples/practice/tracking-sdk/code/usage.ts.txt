// --- 1. 点击：事件委托，一个监听器管全站 ---

document.addEventListener(
  'click',
  (e) => {
    //! closest 向上找最近的 data-track，不用给每个元素单独绑定
    const el = (e.target as HTMLElement).closest<HTMLElement>('[data-track]')
    if (!el) return
    track('click', {
      name: el.dataset.track,
      ...parseExtra(el.dataset.extra), // "position=top&source=feed"
    })
  },
  //! 捕获阶段：避免业务代码 stopPropagation 把埋点拦掉
  true,
)

// --- 2. 曝光：IntersectionObserver + 去重 ---

const exposed = new WeakSet<Element>()

const io = new IntersectionObserver(
  (entries) => {
    for (const entry of entries) {
      if (!entry.isIntersecting) continue
      const el = entry.target as HTMLElement

      //! 同一个元素反复进出视口只算一次，否则数据会爆炸
      if (exposed.has(el)) continue
      exposed.add(el)

      track('exposure', { name: el.dataset.track })
      io.unobserve(el)
    }
  },
  {
    threshold: 0.6, //! 露出 60% 才算曝光
    // rootMargin: '0px 0px -10% 0px',  // 提前/延后触发
  },
)

// --- 3. 批量队列 ---

const queue: TrackEvent[] = []
const BATCH_SIZE = 5
const FLUSH_MS = 2000

function track(type: string, data: Record<string, unknown>): void {
  queue.push({ type, ...data, ts: Date.now() })
  if (queue.length >= BATCH_SIZE) flush()
}

setInterval(() => queue.length && flush(), FLUSH_MS)

// --- 4. 上报：sendBeacon 优先 ---

function flush(): void {
  if (!queue.length) return
  const events = queue.splice(0)
  const body = new Blob([JSON.stringify({ events })], { type: 'application/json' })

  //! sendBeacon：页面关闭/跳转时也能发出，不阻塞导航
  const ok = navigator.sendBeacon('/collect', body)
  if (!ok) {
    // 发送失败（队列满/网络问题）：退化为 fetch keepalive
    void fetch('/collect', {
      method: 'POST',
      body,
      keepalive: true, //! 让请求在页面卸载后继续
    }).catch(() => {
      //! 最终失败：写 localStorage，下次启动补传
      queue.unshift(...events)
    })
  }
}
