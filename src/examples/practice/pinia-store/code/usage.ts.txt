// --- 1. setup store（推荐） ---

// stores/demo.ts
import { defineStore } from 'pinia'
import { computed, ref } from 'vue'

export const useDemoStore = defineStore('demo', () => {
  // state
  const count = ref(0)
  const user = ref('张三')
  const todos = ref<string[]>([])

  // getters
  const double = computed(() => count.value * 2)

  // actions
  function increment(): void {
    count.value++
  }

  //! setup store 必须 return 所有要暴露的 state/getter/action
  return { count, user, todos, double, increment }
})

// --- 2. 持久化：用插件，不要手写 ---

// npm i pinia-plugin-persistedstate
// main.ts
// app.use(createPinia().use(piniaPluginPersistedstate))

// 真实项目里两种写法二选一；这里用不同 id 并列展示，避免同名 store 冲突
export const usePersistedDemoStore = defineStore(
  'demo-persist',
  () => {
    const count = ref(0)
    return { count }
  },
  {
    //! 插件会自动 deep watch + localStorage 读写（等价于本例 20 行手写）
    persist: {
      key: 'moyu-demo-store',
      // 需要只持久化部分字段时用 paths
      // paths: ['count'],
    },
  },
)

// --- 3. 组件里使用 ---

const store = useDemoStore()
store.increment()

//! 解构会丢响应式：必须 storeToRefs（状态用 toRefs 语义，方法直接解构）
import { storeToRefs } from 'pinia'

const { count, user } = storeToRefs(store)
const { increment } = store
