//! npx vite build --mode production
//! 看输出的 chunk 列表：哪个 chunk 超过 500KB，先查它
//! 可视化：rollup-plugin-visualizer 生成 treemap，一眼看到大头

import { visualizer } from 'rollup-plugin-visualizer'

export default defineConfig({
  plugins: [vue(), visualizer({ gzipSize: true, open: true })],
})

// --- 2. 路由懒加载：最有效的一刀 ---

const routes = [
  //! () => import() 让这个组件独立成 chunk，不进首屏
  { path: '/cases', component: () => import('./views/CasesView.vue') },
  { path: '/cases/:category/:id', component: () => import('./views/CasePage.vue') },
]

// --- 3. vendor 拆分：缓存友好 ---

// Vite 8 底层是 Rolldown：manualChunks 会被转成单个 codeSplitting 组，
// 且默认递归收编依赖，容易把 Vue 运行时卷进别的库。
//! 用 codeSplitting.groups + priority：Vue 优先级最高，先被认领
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        codeSplitting: {
          groups: [
            {
              name: 'vendor-vue',
              test: /node_modules[\\/](?:vue|vue-router|@vue)[\\/]/,
              priority: 100,
            },
            {
              name: 'vendor-markdown',
              test: /node_modules[\\/](?:highlight\.js[\\/]lib|marked)[\\/]/,
              priority: 50,
            },
            {
              name: 'vendor-naive-ui',
              test: /node_modules[\\/]naive-ui[\\/]/,
              priority: 40,
              //! 递归收编依赖：naive 依赖 css-render / vooks / lodash 等
              //! 独立小包，必须一起进本 chunk；关掉会被内联进各 Demo
              //! chunk，与 vendor 形成跨 chunk 循环 import，生产构建挂
            },
          ],
        },
      },
    },
  },
})

// Vite 7 及以前（Rollup）仍可用 output.manualChunks：
// output: { manualChunks: { vue: ['vue', 'vue-router'] } }

// --- 4. 按需引入：别整包 import ---

import hljs from 'highlight.js/lib/core'
import typescript from 'highlight.js/lib/languages/typescript'
import xml from 'highlight.js/lib/languages/xml'
hljs.registerLanguage('typescript', typescript)
hljs.registerLanguage('xml', xml)

// --- 7. 什么时候别拆 ---

//! 1. chunk 太小（<10KB）：多一个请求反而更慢，HTTP/2 也有开销
//! 2. 首屏马上要用：拆出去还得立刻加载，没意义
//! 3. 拆得太碎：瀑布请求链变长，弱网更慢
//! 4. 只拆 JS 不管 CSS：样式也会跟着 chunk 走，注意 FOUC

// --- 踩坑清单 ---

//! 2. 分组 test 写死包名，库改名后就失效：用 node_modules 路径正则更稳
//! 3. 拆 vendor 后循环依赖可能报警告：检查 chunk 之间的引用关系
//! 4. CDN 外置要配 external + 全局变量，还要处理 SSR/本地开发
