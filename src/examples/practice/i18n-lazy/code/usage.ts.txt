// --- 1. 语言包拆文件，动态 import 才分包 ---

// locales/zh.ts  locales/en.ts  locales/ja.ts
//! 静态 import 会把所有语言打进首屏 bundle
import zh from './locales/zh'

//! 动态 import 生成独立 chunk，用到才下载
const loaders = {
  en: () => import('./locales/en'),
  ja: () => import('./locales/ja'),
}

// --- 2. 加载状态 + 内存缓存 ---

const cache = new Map<string, Record<string, string>>([['zh', zh]])

async function loadLocale(lang: string): Promise<Record<string, string>> {
  const hit = cache.get(lang)
  //! 已加载过就别再请求（切来切去不应该重复下载）
  if (hit) return hit

  const mod = await loaders[lang as 'en' | 'ja']()
  cache.set(lang, mod.default)
  return mod.default
}

// --- 4. 首屏语言怎么定 ---

//! 优先级：URL 前缀 > localStorage 上次选择 > navigator.language > 默认 zh
const saved = localStorage.getItem('locale')
const lang = saved ?? navigator.language.split('-')[0] ?? 'zh'
