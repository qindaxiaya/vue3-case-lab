// --- 1. 校验：先拦掉，别等传完才报错 ---

function validate(file: File): string | null {
  const allowed = file.type.startsWith('image/') || file.type === 'application/pdf'

  //! 服务端必须再校验一次：前端校验只是体验优化，能被绕过
  if (!allowed) return `${file.name}：只支持图片或 PDF`
  if (file.size > 2 * 1024 * 1024) return `${file.name}：超过 2MB`
  return null
}

// --- 2. 真实上传：XHR 才能拿到进度 ---

function upload(file: File, onProgress: (pct: number) => void): Promise<string> {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest()
    const form = new FormData()
    form.append('file', file)

    //! fetch 没有上传进度事件（只有下载流的 reader），进度条只能用 XHR
    xhr.upload.onprogress = (e) => {
      if (e.lengthComputable) onProgress(Math.round((e.loaded / e.total) * 100))
    }

    xhr.onload = () => {
      if (xhr.status >= 200 && xhr.status < 300) resolve(xhr.responseText)
      else reject(new Error(`HTTP ${xhr.status}`))
    }
    xhr.onerror = () => reject(new Error('网络错误'))
    xhr.onabort = () => reject(new Error('canceled'))

    xhr.open('POST', '/api/upload')
    xhr.send(form)

    //! 暴露取消方法给调用方
    ;(xhr as XMLHttpRequest & { cancel?: () => void }).cancel = () => xhr.abort()
  })
}

// --- 3. 并发队列：别一次发 50 个请求 ---

const CONCURRENCY = 2
const queue: File[] = []
let running = 0

function next(): void {
  while (running < CONCURRENCY && queue.length) {
    const file = queue.shift()!
    running++

    upload(file, (pct) => updateProgress(file, pct))
      .then(() => markSuccess(file))
      .catch((err) => markFailed(file, err.message))
      .finally(() => {
        running--
        //! 一个结束就补位，保持并发数稳定
        next()
      })
  }
}

//! 重试只把 error 改回 pending，不能重置已成功的
//! 取消后要清定时器 / abort 请求，否则回调还会改状态

//! 2. 分片上传：大文件切片 + 并发 + 续传（见 chunk-upload 案例）
//! 3. 图片压缩：canvas 缩小后再传，移动端能省 80% 流量
