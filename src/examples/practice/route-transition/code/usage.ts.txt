// --- 1. 用路由深度判断前进 / 后退 ---

// router/index.ts
router.afterEach((to, from) => {
  //! to.meta.depth / from.meta.depth 由路由表手动标注（0 首页、1 列表、2 详情）
  //! 或用 history.state.position 比较（Vue Router 4 会在 state 里写 position）
  const toDepth = to.meta.depth ?? 0
  const fromDepth = from.meta.depth ?? 0
  direction.value = toDepth >= fromDepth ? 'forward' : 'back'
})
