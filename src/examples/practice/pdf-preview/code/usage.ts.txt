//! </object>

// --- 3. 查看器参数 ---

//! #page=2        跳到第 2 页
//! #zoom=80       缩放 80%（也支持 zoom=page-fit）
//! #toolbar=0     隐藏工具栏（部分浏览器支持）
//! #view=FitH     宽度适配
// 注意：这些参数由「浏览器内置查看器」解析，Chrome/Edge/Firefox 支持度不同

// --- 4. 移动端问题 ---

//! iOS Safari 内嵌 PDF 经常只显示第一页，或者直接弹下载
//! 需要跨端一致时，用 pdfjs-dist 自己渲染到 <canvas>

import * as pdfjs from 'pdfjs-dist'
import workerUrl from 'pdfjs-dist/build/pdf.worker.min.mjs?url'

pdfjs.GlobalWorkerOptions.workerSrc = workerUrl

const doc = await pdfjs.getDocument({ url }).promise
const pdfPage = await doc.getPage(1)
const viewport = pdfPage.getViewport({ scale: 1.5 })
const canvas = document.createElement('canvas')
canvas.width = viewport.width
canvas.height = viewport.height
await pdfPage.render({ canvasContext: canvas.getContext('2d')!, viewport }).promise

// --- 5. 真实项目选型 ---

//! - 只是「能看」：iframe + object URL，10 行代码
//! - 要自定义 UI（翻页/缩放/搜索/高亮/水印）：pdfjs-dist
//! - 要生成 PDF（合同/报表）：pdf-lib（纯前端，支持嵌入中文字体）
//! - 要「禁止下载/加水印/跨端一致」：服务端转图或专业服务
