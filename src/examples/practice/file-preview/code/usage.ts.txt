// --- 1. 判断类型：优先 MIME，再退回扩展名 ---

function kindOf(file: File): 'image' | 'text' | 'pdf' | 'audio' | 'video' | 'other' {
  const type = file.type.toLowerCase()

  if (type.startsWith('image/')) return 'image'
  if (type.startsWith('audio/')) return 'audio'
  if (type.startsWith('video/')) return 'video'
  //! 有些系统给 .md / .log 的 MIME 是空的，只能看扩展名
  if (type === 'application/pdf' || file.name.endsWith('.pdf')) return 'pdf'
  if (type.startsWith('text/') || /json|xml|csv|markdown/.test(type)) return 'text'

  return 'other'
}

// --- 2. 非文本：一个 Blob URL 喂给原生组件 ---

const url = URL.createObjectURL(file)

// --- 3. 文本：直接读内容 ---

const text = await file.text()
//! 大文件要截断，别把 20MB 日志塞进 <pre>
const preview = text.slice(0, 6000)

// --- 4. 切换文件必须释放 ---

function release(old: string | null): void {
  //! ObjectURL 不会自动回收，页面开久了内存暴涨
  if (old) URL.revokeObjectURL(old)
}

// --- 5. PDF 额外参数 ---

//! #page=2&zoom=80：浏览器内置 PDF 查看器支持的部分参数
