//! 第二个泛型是绑定值类型（binding.value）
const vPermission: Directive<HTMLElement, string[]> = {
  // 挂载：首次应用
  mounted(el, binding) {
    apply(el, binding.value)
  },
  // 更新：组件重渲染 / 绑定值变化时重新应用
  updated(el, binding) {
    apply(el, binding.value)
  },
}

function apply(el: HTMLElement, allowed: string[]): void {
  //! 用 display 控制显隐（不销毁元素，适合按钮级权限）
  el.style.display = allowed.includes(currentRole.value) ? '' : 'none'
}

// --- 2. v-lazy：图片懒加载 ---

const ioMap = new WeakMap<HTMLElement, IntersectionObserver>()

const vLazy: Directive<HTMLElement, string> = {
  mounted(el, binding) {
    const io = new IntersectionObserver(
      (entries) => {
        if (entries[0]?.isIntersecting) {
          el.setAttribute('src', binding.value)
          //! 加载后立即断开，避免重复触发
          io.disconnect()
          ioMap.delete(el)
        }
      },
      // 提前 80px 预加载
      { rootMargin: '80px' },
    )
    io.observe(el)
    ioMap.set(el, io)
  },
  //! 卸载时必须 disconnect，否则观察器泄漏
  unmounted(el) {
    ioMap.get(el)?.disconnect()
    ioMap.delete(el)
  },
}

// --- 3. v-click-outside：点外部关闭 ---

const handlerMap = new WeakMap<HTMLElement, (e: PointerEvent) => void>()

const vClickOutside: Directive<HTMLElement, () => void> = {
  mounted(el, binding) {
    const handler = (e: PointerEvent): void => {
      //! contains 判断点击是否落在元素内部（含子节点）
      if (!el.contains(e.target as Node)) binding.value()
    }
    handlerMap.set(el, handler)
    // 捕获阶段监听，避免被内部 stopPropagation 破坏
    document.addEventListener('pointerdown', handler, true)
  },
  unmounted(el) {
    const handler = handlerMap.get(el)
    if (handler) document.removeEventListener('pointerdown', handler, true)
    handlerMap.delete(el)
  },
}
