const AsyncChart = defineAsyncComponent(() => import('./HeavyChart.vue'))

// --- 步骤二：失败重试的标准姿势（换 key 重建） ---

const attempt = ref(0)

//! 重试不靠 retry() API（defineAsyncComponent 没有）——换 :key 强制销毁重建，
//! 新实例重新走一遍 loader。这是社区统一做法。
function retry(): void {
  attempt.value++
}
