// --- 现场 1：解构 reactive ---

const state = reactive({ count: 0 })

//! 错误：拿到的是「当时的数字」，与响应式系统断开
const { count } = state

//! 正确：toRef / toRefs 建立只读连接
const countRef = toRef(state, 'count')
// 或：const { count } = toRefs(state)

state.count++
console.log(count) // 0（快照不动）
console.log(countRef.value) // 1（连接跟着走）

// --- 现场 2：整体重新赋值 ---

let form = reactive({ name: 'a' })

//! 错误：变量被换成裸对象，旧 Proxy 的依赖永远等不到更新
form = { name: 'b' }

//! 正确：原地改，保持同一个 Proxy
Object.assign(form, { name: 'b' })
// 或者从头就用 ref：const form = ref({ name: 'a' }); form.value = { name: 'b' }

// --- 现场 3：展开浅拷贝 ---

//! 错误：{ ...state } 拷走的是值，不是响应式连接
const copy = { ...state }

//! 正确：直接读源对象，或明确这是不响应的快照
const n = state.count

// --- 现场 4：解构 props ---

// 子组件里：
// const props = defineProps<{ msg: string }>()
//! 错误：setup 只跑一次，msg 是首帧快照
// const { msg } = props
//! 正确：toRef 保持实时读取
// const msg = toRef(props, 'msg')
//! Vue 3.5+ 解构语法糖（编译期自动 toRef）
// const { msg } = defineProps<{ msg: string }>()

// --- 现场 5：Vue 3 已修复的老坑 ---

const arr = reactive([1, 2, 3])
const obj = reactive<{ late?: number }>({})

//! Vue 3 用 Proxy 拦截：下标赋值 / 新增属性都直接生效
arr[0] = 99 // Vue 2 要 Vue.set(arr, 0, 99)
obj.late = 1 // Vue 2 要 Vue.set(obj, 'late', 1)
