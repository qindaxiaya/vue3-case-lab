const price = ref(120)
const qty = ref(2)
const noise = ref(0) // 无关状态：只影响渲染

// --- 1. computed：依赖不变就吃缓存 ---

//! computed 是惰性的：第一次读取才求值，之后依赖不变直接返回缓存
const total = computed(() => price.value * qty.value)

// --- 2. 模板里调方法：没有缓存 ---

// 模板：{{ methodTotal() }}
//! 每次组件重新渲染都会调用一次——即使 price/qty 根本没变
function methodTotal(): number {
  return price.value * qty.value
}

// --- 3. watch：只在依赖变化时触发 ---

//! watch 不产生值，适合做副作用（请求、日志、写 localStorage）
watch([price, qty], () => {
  console.log('价格或数量变了')
})

// --- 为什么模板里别写方法 ---

// 一个渲染依赖很多状态：输入框光标、主题、父组件传参……
// 任一变化都会重渲染 → 方法全部重算；computed 则按依赖精确失效。
//
// 什么时候可以用方法：
// - 只在小列表/轻量计算里
// - 需要每次强制拿最新值（例如随机数、时间）

// --- 什么时候不该用 computed ---

//! computed 里不要改状态、不要发请求——它是「纯计算 + 缓存」，副作用用 watch
//! 不要为了“缓存”把事件处理也包成 computed（事件不是派生值）
