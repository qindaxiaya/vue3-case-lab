// 长列表时间切片教学片段
// 重点行以 //! 标记——展示时高亮、复制时自动去掉标记
// ============================================================

import { nextTick, ref } from 'vue'

interface Row {
  id: number
  text: string
}

const totalCount = ref(30000)
const batchSize = ref(300)
const rows = ref<Row[]>([])
const firstPaintMs = ref<number | null>(null)
const maxBlockMs = ref<number | null>(null)
const totalMs = ref<number | null>(null)
const frames = ref(0)
const clickLatency = ref<number | null>(null)

let rafId = 0
let runToken = 0

function makeRows(count: number, start: number): Row[] {
  return Array.from({ length: count }, (_, i) => ({
    id: start + i,
    text: `数据行 #${start + i}`,
  }))
}

//! 双 rAF：等浏览器真正完成一次绘制，再记首帧
function nextFrame(): Promise<void> {
  return new Promise((resolve) => {
    requestAnimationFrame(() => requestAnimationFrame(() => resolve()))
  })
}

// --- 1. 一次性全量：长任务 ---

async function renderFull(): Promise<void> {
  const token = ++runToken
  const t0 = performance.now()

  //! 一次赋值几万条：Vue 需要一次性创建几万个 vnode + DOM
  rows.value = makeRows(totalCount.value, 0)
  await nextTick()
  if (token !== runToken) return

  // 主线程真正被占住的时间
  maxBlockMs.value = performance.now() - t0
  await nextFrame()
  if (token !== runToken) return

  firstPaintMs.value = performance.now() - t0
  totalMs.value = maxBlockMs.value
}

// --- 2. 每帧一批：把长任务切碎 ---

function renderSlice(): void {
  const token = ++runToken
  const t0 = performance.now()
  let produced = 0

  const step = async (): Promise<void> => {
    if (token !== runToken) return

    //! 每批单独计时：push + nextTick 里的 DOM 更新就是这一批占住主线程的时间
    const batchStart = performance.now()
    const batch = makeRows(
      Math.min(batchSize.value, totalCount.value - produced),
      produced,
    )
    rows.value.push(...batch)
    produced += batch.length
    frames.value++

    await nextTick()
    if (token !== runToken) return

    const block = performance.now() - batchStart
    maxBlockMs.value =
      maxBlockMs.value === null ? block : Math.max(maxBlockMs.value, block)

    if (firstPaintMs.value === null) {
      await nextFrame()
      if (token !== runToken) return
      firstPaintMs.value = performance.now() - t0
    }

    //! 只处理一批就交还主线程：浏览器有机会渲染、响应点击
    if (produced < totalCount.value) {
      rafId = requestAnimationFrame(step)
    } else {
      totalMs.value = performance.now() - t0
    }
  }

  rafId = requestAnimationFrame(step)
}

// --- 3. 点击响应：从 event.timeStamp 起算 ---

function ping(event: MouseEvent): void {
  //! 主线程被长任务占住时，处理函数本身就会晚执行；
  //! 从 event.timeStamp 起算才能把「排队等待」算进去
  const start = event.timeStamp > 0 ? event.timeStamp : performance.now()
  setTimeout(() => {
    clickLatency.value = Math.round(performance.now() - start)
  }, 0)
}

// --- 4. requestIdleCallback：更省主线程，但时机不可控 ---

function renderIdle(): void {
  let produced = 0
  const step = (deadline: IdleDeadline): void => {
    while (produced < totalCount.value && deadline.timeRemaining() > 2) {
      const batch = makeRows(Math.min(batchSize.value, totalCount.value - produced), produced)
      rows.value.push(...batch)
      produced += batch.length
    }
    if (produced < totalCount.value) requestIdleCallback(step)
  }
  requestIdleCallback(step)
}

// ---------- 实践建议 ----------

// 1. 长列表优先虚拟滚动；确实要一次展示全部数据时，再用时间切片。
// 2. 每批数量按实测最大阻塞调：批次越大单次阻塞越长，批次数越少总耗时越短。
// 3. 首帧用 nextTick + 双 rAF，别在 push 后立刻读时间。
// 4. 点击响应用 event.timeStamp 起算，才能把主线程排队时间算进去。
// 5. 重复运行 / 卸载要取消 rAF，并用运行令牌丢弃过期批次。
// 6. 切片只解决主线程阻塞，不解决 DOM 数量与内存占用。
