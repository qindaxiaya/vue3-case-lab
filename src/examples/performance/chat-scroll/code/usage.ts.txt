// --- 1. 判底：距离小于阈值才算「在底部」 ---

function isNearBottom(el: HTMLElement): boolean {
  //! 不要用 === 0：像素小数、滚动惯性都会差几像素
  return el.scrollHeight - el.scrollTop - el.clientHeight < 80
}

function onScroll(): void {
  stickToBottom.value = isNearBottom(scroller.value!)
  //! 滑回底部要清未读
  if (stickToBottom.value) unread.value = 0
}

// --- 2. 追加消息：底部跟随 / 上翻不打扰 ---

async function append(msg: Msg): Promise<void> {
  messages.value.push(msg)
  await nextTick() // 等 DOM 高度更新

  if (stickToBottom.value) {
    scroller.value!.scrollTop = scroller.value!.scrollHeight
  } else {
    //! 上翻时千万别改 scrollTop，否则用户正在看的内容被拽走
    unread.value++
  }
}

// --- 3. 内容高度变化也要吸底 ---

// 图片加载、折叠展开、字体切换都会改变 scrollHeight
//! 用 ResizeObserver 观察内容容器，而不是只监听消息数组变化
const ro = new ResizeObserver(() => {
  if (stickToBottom.value) scroller.value!.scrollTop = scroller.value!.scrollHeight
})
ro.observe(contentEl.value!)

// --- 4. 上翻加载历史：保持视线位置 ---

async function loadHistory(): Promise<void> {
  const el = scroller.value!
  const prevHeight = el.scrollHeight

  messages.value.unshift(...olderMessages)
  await nextTick()

  //! 前插后补差：scrollTop += 新增高度，视觉上停在原处
  el.scrollTop += el.scrollHeight - prevHeight
}
