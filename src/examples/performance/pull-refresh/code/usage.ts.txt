// --- 1. 只在顶部允许下拉 ---

function onPointerDown(e: PointerEvent): void {
  //! 列表已经滚下去了就别再接管手势，否则用户滚不动
  if (scroller.value!.scrollTop > 0) return
  startY = e.clientY
  dragging.value = true
  ;(e.currentTarget as HTMLElement).setPointerCapture(e.pointerId) // 手指移出元素也能收到事件
}

// --- 2. 阻尼：位移永远小于手指 ---

function onPointerMove(e: PointerEvent): void {
  if (!dragging.value) return
  const dy = e.clientY - startY
  //! dy * 0.45：手指拉 100px，内容只下移 45px；上限 110px 防止越拉越远
  pull.value = Math.max(0, Math.min(110, dy * 0.45))
}

// --- 3. 松手：超过阈值才刷新 ---

async function onPointerUp(): Promise<void> {
  dragging.value = false
  if (pull.value >= THRESHOLD) {
    refreshing.value = true
    pull.value = THRESHOLD // 停在阈值处显示 loading，别弹回去
    await fetchData()
    refreshing.value = false
  }
  //! 未达阈值：pull 归零，CSS transition 自动回弹
  pull.value = 0
}

// --- 4. 触底加载：用「距底距离」判定 ---

function onScroll(): void {
  const el = scroller.value!
  //! scrollHeight - scrollTop - clientHeight 才是距底距离（不要用 scrollTop 直接比）
  const nearBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 48
  if (nearBottom && !loading.value && !finished.value) loadMore()
}
