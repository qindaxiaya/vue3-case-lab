// ============================================================
// 往返动画：rAF 循环 + 正弦缓动（track 演示对应代码）
// 重点行用 //! 标记——展示时高亮、复制时自动去标记
// ============================================================

// --- 步骤一：rAF 基本配对（启动 / 取消必须成对） ---

/** rAF 返回的句柄：取消循环全靠它 */
let rafId = 0

/** 循环是否在跑（暂停后不再续订，循环自然终止） */
let running = false

function loop(now: number) {
  render(now)

  //! 续订下一帧：running 为 false 时不再排队，循环自行终止
  if (running) {
    rafId = requestAnimationFrame(loop)
  }
}

function start() {
  if (running) return
  running = true
  //! 启动：把 loop 交给浏览器（回调参数 now = performance.now() 时间戳）
  rafId = requestAnimationFrame(loop)
}

function stop() {
  running = false
  //! 取消：与 requestAnimationFrame 成对；只停调度不取消会泄漏句柄
  cancelAnimationFrame(rafId)
}

// --- 步骤二：时间戳驱动 + 正弦缓动往返 ---

const TRACK = 320 // 轨道长 px
const PERIOD = 3.2 // 往返周期秒
let startTs = 0 // 本段循环起点时间戳

function render(now: number) {
  const elapsed = (now - startTs) / 1000
  const phase = (elapsed % PERIOD) / PERIOD // 归一化 0..1

  //! 正弦缓动：sin 在两端斜率为 0 → 速度自然归零，折返无突变
  //! （线性三角波在折返点速度瞬间反向，视觉上「生硬」）
  const wave = (Math.sin(phase * Math.PI * 2 - Math.PI / 2) + 1) / 2 // 0..1

  // transform 驱动位移：只触发合成，不触发 layout/paint
  box.style.transform = `translateX(${wave * TRACK}px)`
}

// 对比 setInterval：与刷新率无关地按时触发，动画会与屏幕刷新错拍（jank）；
// rAF 只在「即将重绘」前执行，天然对齐刷新节奏，后台标签自动节流。

// --- 步骤三：visibilitychange 暂停（可选优化） ---

//! 页面不可见时停循环（后台标签 rAF 本就会被浏览器降频，主动停更省电）
document.addEventListener('visibilitychange', () => {
  if (document.hidden) {
    stop()
  } else {
    start() // 恢复时注意平移 startTs，避免瞬间跳变
  }
})
