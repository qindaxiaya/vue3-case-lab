// --- 1. 消费 SSE：边读边解析 ---

async function consumeSSE(url: string): Promise<void> {
  const res = await fetch(url, { signal: controller.signal })
  const reader = res.body!.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  while (true) {
    const { done, value } = await reader.read()
    if (done) break

    buffer += decoder.decode(value, { stream: true })

    //! SSE 以空行分隔事件；最后一段可能不完整，留在 buffer 里等下一块
    const parts = buffer.split('\n\n')
    buffer = parts.pop() ?? ''

    for (const part of parts) {
      for (const line of part.split('\n')) {
        if (!line.startsWith('data:')) continue
        //! data 后面的第一个空格是格式要求，要去掉
        const text = line.slice(5).trimStart()
        received.value += text // 进缓冲区，不直接上屏
        backlog.value += text.length
      }
    }
  }
}

// --- 2. rAF 匀速消费 ---

function startConsumer(): void {
  const tick = (): void => {
    if (backlog.value > 0) {
      //! 每帧固定预算：到达再快，屏幕推进速度稳定
      const take = Math.min(3, backlog.value)
      shownLen.value += take
      backlog.value -= take
    }
    rafId = requestAnimationFrame(tick)
  }
  rafId = requestAnimationFrame(tick)
}

// --- 3. 更聪明的自适应消费 ---

//! 积压多时每帧多输出一点（追帧），积压少时保持 1 字
//! 但别无限追：屏幕刷太快人眼也读不过来，且 markdown 重渲染会卡
const take = Math.min(
  backlog.value,
  Math.max(1, Math.ceil(backlog.value / 6)), // 最多「六分之一库存」
  6,
)
