// --- 1. 用 fetch 读 SSE（支持 POST / 自定义头） ---

const controller = new AbortController()

async function streamChat(prompt: string): Promise<void> {
  const res = await fetch('/api/chat/stream', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ prompt }),
    signal: controller.signal, //! 用户点「停止」时 abort
  })

  const reader = res.body!.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  while (true) {
    const { done, value } = await reader.read()
    if (done) break

    //! stream: true 保证多字节中文不会被截断
    buffer += decoder.decode(value, { stream: true })

    //! SSE 事件以空行分隔；不完整的最后一段留在 buffer
    const chunks = buffer.split('\n\n')
    buffer = chunks.pop() ?? ''

    for (const chunk of chunks) {
      let id = ''
      let data = ''

      for (const line of chunk.split('\n')) {
        if (line.startsWith('id:')) id = line.slice(3).trim()
        else if (line.startsWith('data:')) data += line.slice(5).trimStart()
      }

      if (id) lastEventId = id //! 记住断点
      if (data) appendToken(data)
    }
  }
}

// --- 2. 断线重连：带 Last-Event-ID ---

async function streamWithRetry(prompt: string): Promise<void> {
  let retries = 0

  while (retries < 3) {
    try {
      await streamChat(prompt) // 请求头里带 Last-Event-ID
      return
    } catch (err) {
      if (controller.signal.aborted) throw err // 用户主动中止，不重连
      retries++

      //! 指数退避：1s、2s、4s，避免服务端刚崩就被打满
      const delay = 1000 * 2 ** (retries - 1)
      await new Promise((r) => setTimeout(r, delay))
    }
  }
}

// --- 3. 发送请求时带上断点 ---

fetch('/api/chat/stream', {
  headers: {
    //! 服务端据此跳过已发送的 token，实现「续传不重复」
    'Last-Event-ID': lastEventId,
  },
})

// --- 4. EventSource：简单但不是万能的 ---

const es = new EventSource('/api/chat/stream')
es.onmessage = (e) => appendToken(e.data)
es.onerror = () => {
  //! 浏览器会自动重连（并自动带上 Last-Event-ID），但：
  //! - 只能 GET，不能带 POST body / 自定义请求头（除了 cookie）
  //! - 无法主动中断某一次请求，只能 close() 整个连接
  //! - 服务端必须返回 text/event-stream，且不能有代理缓冲
}
es.close()
