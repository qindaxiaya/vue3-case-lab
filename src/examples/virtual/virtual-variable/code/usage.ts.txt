// --- 1. 为什么定高公式失效 ---

// 定高：start = Math.floor(scrollTop / rowHeight)
//! 不定高：每行高度不同，只能靠「前缀和 + 二分」定位

// --- 2. 前缀和：offsets[i] = 前 i 行总高度 ---

const rows: Msg[] = [
  { height: 44 },
  { height: 92 },
  { height: 60 },
]

const offsets: number[] = [0]
for (const row of rows) {
  //! offsets[i] 就是第 i 行的顶部位置
  offsets.push(offsets[offsets.length - 1] + row.height)
}

// --- 3. 二分查找起始行 ---

function findStart(scrollTop: number): number {
  let lo = 0
  let hi = rows.length - 1

  while (lo < hi) {
    const mid = (lo + hi) >> 1
    //! offsets[mid + 1] 是第 mid 行的底边；底边 <= scrollTop 说明整行在视口上方
    if (offsets[mid + 1] <= scrollTop) lo = mid + 1
    else hi = mid
  }
  return lo
}
