// ============================================================
// clamp 流体字号教学片段
// 重点行用 //! 标记——展示时高亮、复制时自动去标记
// ============================================================

/* ---------- 1. 流体字号：一条声明走天下 ---------- */

.fluid-title {
  //! clamp(最小 1rem = 16px，相对值 8cqi = 容器宽的 8%，最大 2.5rem = 40px)
  //! 相对值随容器宽度连续变化，clamp 只做三值收敛——产出从一开始
  //! 就是一条连续曲线，断点表整张作废
  font-size: clamp(1rem, 8cqi, 2.5rem);
}

//! 被驱动对象必须先成为容器：没有 inline-size，cqi 会退回按视口算
.title-host {
  container-type: inline-size;
}

/* 正文按标题比例缩：跟着同一条流体曲线走，不会出现「标题缩了正文 HUGE」 */
.fluid-body {
  font-size: 0.55em; /* 1em = 容器根字号，与标题同源 */
}

/* ---------- 2. 旧做法对照：断点表硬档位 ---------- */

//! 每一档都是一个手工挑选的固定数字；
//! 窗口滑过档位边界时字号瞬间跳变，档位之间的宽度全被浪费
.title-host {
  font-size: 16px;
}

@media (min-width: 360px) {
  .title-host { font-size: 23px; }
}

@media (min-width: 480px) {
  .title-host { font-size: 28px; }
}

@media (min-width: 620px) {
  .title-host { font-size: 34px; }
}

@media (min-width: 760px) {
  .title-host { font-size: 40px; }
}
