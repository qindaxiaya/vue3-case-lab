/* ---------- 1. 给宿主注册为「查询容器」 ---------- */

.card-host {
  //! container-type: inline-size 让元素成为最近查询容器，
  //! 同时建立独立的包含块（其内部布局变化不会外溢打扰页面其他区域）
  container-type: inline-size;
}

/* ---------- 2. 容器查询：按容器宽度切换排向 ---------- */

/* 默认基线：窄容器里纵排，缩略图整行 */
.cq-card {
  display: flex;
  flex-direction: column;
  gap: 0.5rem;
}

//! @container 对「最近祖先容器」逐个求值：同一份规则，
//! 宽容器 / 窄容器 / 并排两列各自独立判定，零复制
@container (min-width: 520px) {
  .cq-card {
    flex-direction: row;   //! 达到 520px 才横排——断点写的是容器宽度
    align-items: center;
  }

  .cq-card .thumb {
    width: 96px; /* 横排时缩略图固定宽，文字占余下空间 */
  }
}

/* ---------- 3. 旧做法对照：媒体查询只跟视口走 ---------- */

//! 视口到了 900px 就全部横排——哪怕卡片正躺在 250px 的窄栏里
//! （组件摆错位置布局就跟着错，且要按每个摆放位置复制调整）
@media (min-width: 900px) {
  .cq-card {
    flex-direction: row;
    align-items: center;
  }

  .cq-card .thumb {
    width: 96px;
  }
}
