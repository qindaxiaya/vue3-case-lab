/* ---------- 1. 占位：比例写在加载之前 ---------- */

.media-grid {
  display: grid;
  grid-template-columns: repeat(auto-fill, minmax(148px, 1fr));
  gap: 0.5rem;
}

.media-cell {
  //! aspect-ratio: 16 / 9 让元素在图片到来之前就有正确高度——
  //! 高度 = 宽度 ÷ 1.7778，由样式直接算出，不等网络
  aspect-ratio: 16 / 9;
  border-radius: 0.25rem;
  background-color: rgba(0, 0, 0, 0.06); /* 低饱和底色当「加载中」皮肤 */
  overflow: hidden;
  min-width: 0;
}

.media-cell img {
  display: block;
  width: 100%;
  height: 100%;      //! 图片只负责「填进早已备好的位置」：object-fit 裁边也稳
  object-fit: cover;
}

/* 对照组（旧写法）：什么都不声明的裸图片 */
.media-cell img.plain {
  display: block;
  width: 100%;
  height: auto;      //! 加载前高度 0，加载后按原图比例撑开——下方内容被顶走
}
