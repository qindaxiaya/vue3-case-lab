// ============================================================
// 抽屉样式片段（与 Demo.vue 的 .dw-mask / .dw-panel 一致）
// 重点行以 //! 标记：展示时高亮，复制时自动去掉标记
// ============================================================

/* ---------- 1. 遮罩：透明度过渡 ---------- */

.dw-mask {
  position: fixed;
  inset: 0;
  z-index: 1200;
  display: flex;
  justify-content: flex-end;
  background-color: rgb(15 23 42 / 0.45);
  //! 初始透明度 0：关闭状态下遮罩不可见，打开时再过渡到 1
  opacity: 0;
  transition: opacity 0.22s var(--ease-out);
}

.dw-mask.shown {
  opacity: 1;
}

/* 旧模式：没有过渡，关闭时直接消失 */
.dw-mask.instant {
  transition: none;
}

/* ---------- 2. 面板：位移进出场 ---------- */

.dw-panel {
  display: flex;
  flex-direction: column;
  width: min(320px, 86vw);
  height: 100%;
  border-left: 1px solid var(--border);
  background-color: var(--surface);
  //! 默认停在屏幕外：位移量是自身宽度的 100%，不同宽度都不用改代码
  transform: translateX(100%);
  transition: transform 0.24s var(--ease-out);
}

.dw-panel.shown {
  transform: translateX(0);
}

.dw-panel.instant {
  transition: none;
}

/* ---------- 3. 退场时的裁切与滚动 ---------- */

.dw-body {
  flex: 1;
  //! 面板自身铺满视口高度，内容多时只让中间区域滚动，头部与底部固定
  overflow-y: auto;
  padding: 0.8rem;
}

/* ---------- 4. 减少动态效果：真实项目应降级，本演示用开关模拟 ---------- */
/* 真实项目：系统开启「减少动态效果」（prefers-reduced-motion: reduce）时
   不做侧滑与淡出，关闭直接切换；本站是教学演示站，不跟随系统降级，
   降级分支由「模拟减少动态效果」开关触发（打开后给元素加 .instant，
   脚本也会跳过等待过渡的步骤，见 logic.ts）。
   旧做法（直接卸载）同样走 .instant，两种关闭方式的差别才看得出来 */
.instant.dw-mask,
.instant.dw-panel {
  //! 不做侧滑与淡出，关闭直接切换
  transition: none;
}
