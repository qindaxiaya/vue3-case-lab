//! 一次点击回调就是一个宏任务；所有异步任务先把它一次排入队列
function onClick(): void {
  // 定时器：下一个宏任务（最快也要约 1ms 后才可能执行）
  setTimeout(() => console.log('4. setTimeout 0ms'), 0)
  setTimeout(() => console.log('5. setTimeout 200ms'), 200)

  // 微任务：排在当前宏任务结束后的最前面
  Promise.resolve().then(() => console.log('2. Promise.then 微任务'))
  queueMicrotask(() => console.log('3. queueMicrotask 微任务'))

  //! 渲染前的逐帧回调：浏览器下一帧绘制前触发，早于页面重绘
  requestAnimationFrame(() => console.log('6. 逐帧回调（渲染前）'))

  //! 主线程忙 200ms 只拖住当前宏任务，微任务会在忙完后立刻插队到定时器前面
  busyWait(200)
}

//! 事件循环一轮的真实顺序：宏任务 → 清空全部微任务 → （到渲染时机）rAF 回调 → 渲染
function busyWait(ms: number): void {
  const end = performance.now() + ms
  while (performance.now() < end) {
    // 空转等待：占住主线程，模拟同步重活
  }
}

// ---------- 实践建议 ----------

// 1. 需要马上但异步（比如把本次状态写入快照）用微任务：Promise.then / queueMicrotask
// 2. 需要下一轮再跑的修补任务才用 setTimeout；0ms 也会插到微任务后面
// 3. 和绘制同频的逻辑（进度条、拖拽跟随）用逐帧回调，别用定时器硬凑
// 注意：微任务里再产生微任务会让队列一直清不完（饿死页面），
//       长链 Promise 不要在同一轮里无限 then 接 then。
